      <p class="description">
        내 주변 맛집과 카페를 찾아보고<br>

        <button id="cafeTab" class="tab-button" type="button" onclick="switchCategory('CE7')">
          ☕ 카페
        </button>

      <!-- 직접 키워드 검색 박스 -->
      <div class="search-box">

      <p id="status" aria-live="polite">

  <script>
    // 카카오 지도 및 관련 객체를 저장할 변수
    let map;
    let placeService;
    let currentInfoWindow = null;
    let currentCenterPosition = null;

    // 카테고리/검색 마커 및 현재 위치 마커를 관리할 배열 및 변수
    let categoryMarkers = [];
    let currentLocationMarker = null;

    // 현재 선택된 카테고리 코드 (기본값: FD6 - 음식점)
    let currentCategoryCode = "FD6";

    // 마커 이미지 URL 정의 (카카오맵 기본 핀 활용)
    const MARKER_IMAGES = {
      // 음식점: 빨간색 핀
      FD6: "https://t1.daumcdn.net/localimg/localimages/07/mapapidoc/markerStar.png",
      // 카페: 파란색 핀 (기본 카카오 핀)
      CE7: "https://t1.daumcdn.net/mapjsapi/images/2x/marker.png",
      // 현재 위치: 특수 표시 마커
      LOCATION: "https://t1.daumcdn.net/localimg/localimages/07/mapapidoc/a110001_08.png"
    };

    // 지도 초기화 함수
    function initializeMap() {
      // 기본 위치 (서울 시청)
      const defaultPosition = new kakao.maps.LatLng(37.5665, 126.9780);
      currentCenterPosition = defaultPosition;

      // 지도 생성
      map = new kakao.maps.Map(document.getElementById("map"), {
        center: defaultPosition,
        level: 4
      });

      // 카카오 장소 검색 객체 생성
      placeService = new kakao.maps.services.Places();

      document.getElementById("status").textContent = "현재 위치를 확인하고 있습니다.";

      // Geolocation을 통한 현재 위치 탐색
      if (!navigator.geolocation) {
        document.getElementById("status").textContent = "위치 기능을 지원하지 않아 서울 시청을 중심으로 탐색합니다.";
        searchCategoryPlaces(currentCategoryCode);
        return;
      }

      navigator.geolocation.getCurrentPosition(
        function (position) {
          // 사용자 위치 좌표 업데이트
          currentCenterPosition = new kakao.maps.LatLng(
            position.coords.latitude,
            position.coords.longitude
          );

          // 지도 중심 이동
          map.setCenter(currentCenterPosition);

          // 현재 위치 마커 표시
          displayCurrentLocationMarker(currentCenterPosition);

          document.getElementById("status").textContent = "현재 위치를 중심으로 주변 장소를 검색합니다.";
          
          // 현재 위치 기반 장소 검색 실행
          searchCategoryPlaces(currentCategoryCode);
        },
        function () {
          // 위치 정보를 얻지 못한 경우 기본 위치(서울 시청) 사용
          document.getElementById("status").textContent = "현재 위치를 확인할 수 없어 서울 시청 주변을 탐색합니다.";
          searchCategoryPlaces(currentCategoryCode);
        },
        {
          enableHighAccuracy: true,
          timeout: 10000,
          maximumAge: 60000
        }
      );
    }

    // 현재 위치 마커 생성 및 표시
    function displayCurrentLocationMarker(position) {
      if (currentLocationMarker) {
        currentLocationMarker.setMap(null);
      }

      const markerSize = new kakao.maps.Size(24, 35);
      const markerImage = new kakao.maps.MarkerImage(MARKER_IMAGES.LOCATION, markerSize);

      currentLocationMarker = new kakao.maps.Marker({
        map: map,
        position: position,
        title: "내 현재 위치",
        image: markerImage
      });
    }

    // 카테고리 탭 변경 시 호출되는 함수
    function switchCategory(categoryCode) {
      currentCategoryCode = categoryCode;

      // 탭 버튼 UI 활성화 상태 전환
      document.getElementById("restaurantTab").classList.toggle("active", categoryCode === "FD6");
      document.getElementById("cafeTab").classList.toggle("active", categoryCode === "CE7");

      // 카테고리 장소 검색 재실행
      searchCategoryPlaces(categoryCode);
    }

    // 카테고리 코드로 주변 장소를 검색하는 함수
    function searchCategoryPlaces(categoryCode) {
      // 기존 마커 및 인포윈도우 지우기
      clearMarkers();

      const options = {
        location: currentCenterPosition,
        radius: 2000, // 반경 2km 이내 검색
        sort: kakao.maps.services.SortBy.DISTANCE
      };

      // 카테고리 코드 검색 API 호출
      placeService.categorySearch(categoryCode, placesSearchCallback, options);
    }

    // 키워드로 장소를 검색하는 함수 (직접 입력 검색)
    function searchPlacesByKeyword() {
      const keyword = document.getElementById("keyword").value.trim();

      if (!keyword) {
        alert("검색어를 입력해 주세요!");
        return;
      }

      clearMarkers();
      document.getElementById("status").textContent = `'${keyword}' 검색 중입니다...`;

      const options = {
        location: currentCenterPosition,
        radius: 5000
      };

      placeService.keywordSearch(keyword, placesSearchCallback, options);
    }

    // 장소 검색 결과 콜백 함수
    function placesSearchCallback(data, status, pagination) {
      if (status === kakao.maps.services.Status.OK) {
        displayPlaces(data);
        document.getElementById("status").textContent = `총 ${data.length}개의 장소를 찾았습니다.`;
      } else if (status === kakao.maps.services.Status.ZERO_RESULT) {
        document.getElementById("status").textContent = "검색 결과가 존재하지 않습니다.";
        document.getElementById("resultList").innerHTML = "";
      } else if (status === kakao.maps.services.Status.ERROR) {
        document.getElementById("status").textContent = "검색 중 오류가 발생했습니다.";
      }
    }

    // 검색된 장소 목록과 마커를 화면에 출력하는 함수
    function displayPlaces(places) {
      const listElement = document.getElementById("resultList");
      listElement.innerHTML = ""; // 기존 리스트 초기화

      const bounds = new kakao.maps.LatLngBounds();

      places.forEach((place, index) => {
        const placePosition = new kakao.maps.LatLng(place.y, place.x);
        
        // 카테고리에 맞는 마커 생성
        const marker = createCategoryMarker(placePosition, currentCategoryCode);
        categoryMarkers.push(marker);

        // 마커 클릭 시 인포윈도우 표시
        kakao.maps.event.addListener(marker, "click", function () {
          openInfoWindow(marker, place);
        });

        // 사이드바 목록 요소 생성
        const listItem = document.createElement("li");
        listItem.className = "place-item";
        
        const addressText = place.road_address_name || place.address_name;
        listItem.innerHTML = `
          <div class="place-name">${place.place_name}</div>
          <div class="place-address">${addressText}</div>
        `;

        // 리스트 클릭 시 지도 위치 이동 및 인포윈도우 표시
        listItem.addEventListener("click", function () {
          map.panTo(placePosition);
          openInfoWindow(marker, place);
        });

        listElement.appendChild(listItem);
        bounds.extend(placePosition);
      });
    }

    // 카테고리별 마커 생성 함수
    function createCategoryMarker(position, categoryCode) {
      const imageUrl = MARKER_IMAGES[categoryCode] || MARKER_IMAGES.CE7;
      const imageSize = new kakao.maps.Size(24, 35);
      const markerImage = new kakao.maps.MarkerImage(imageUrl, imageSize);

      return new kakao.maps.Marker({
        map: map,
        position: position,
        image: markerImage
      });
    }

    // 마커/목록 클릭 시 인포윈도우(팝업)를 띄우는 함수
    function openInfoWindow(marker, place) {
      if (currentInfoWindow) {
        currentInfoWindow.close();
      }

      const addressText = place.road_address_name || place.address_name;
      const content = `
        <div class="info-window">
          <strong>${place.place_name}</strong>
          <span>${addressText}</span>
        </div>
      `;

      currentInfoWindow = new kakao.maps.InfoWindow({
        content: content,
        removable: true
      });

      currentInfoWindow.open(map, marker);
    }

    // 기존 마커들과 인포윈도우를 지우는 함수
    function clearMarkers() {
      // 이전 인포윈도우 닫기
      if (currentInfoWindow) {
        currentInfoWindow.close();
        currentInfoWindow = null;
      }

      // 이전 카테고리 마커들 제거
      categoryMarkers.forEach(marker => marker.setMap(null));
      categoryMarkers = [];
    }

    // SDK 준비 완료 시 지도 초기화 실행
    window.addEventListener("load", function () {
      if (typeof kakao === "undefined" || !kakao.maps) {
        document.getElementById("status").textContent =
          "지도 로딩에 실패했습니다. JavaScript 키와 도메인 설정을 확인하세요.";
        return;
      }

      kakao.maps.load(function () {
        initializeMap();
      });
    });

    // 검색 버튼 클릭 이벤트 등록
    document.getElementById("searchButton").addEventListener("click", searchPlacesByKeyword);

    // 검색창 엔터키 입력 이벤트 등록
    document.getElementById("keyword").addEventListener("keypress", function (e) {
      if (e.key === "Enter") {
        searchPlacesByKeyword();
      }
    });
  </script>
